# --- BLOQUE 0: reset con datos limpios (2 min) ---------------------------------
cd ~/Facultad/3er_ano/paradigmas_III/TFinder_AE2/TFinder

# --- BLOQUE 1: helpers, servicios, frontend y tokens (2 min) --------------------
tok() { curl -s -X POST "localhost:8001/api/v1/auth/$1" -H 'Content-Type: application/json' -d "$2" | jq -r '.access_token // ""'; }
alta() { local e="$1" n="$2" t; t=$(tok login "{\"email\":\"$e\",\"password\":\"DemoTFinder123!\"}"); [ -n "$t" ] || t=$(tok register "{\"email\":\"$e\",\"password\":\"DemoTFinder123!\",\"nombre\":\"$n\"}"); printf '%s' "$t"; }
cuenta() { docker compose exec -T postgres psql -U tfinder -d mesas_db -tAc "select id||'  '||email from usuarios where email='$1'"; }
curl -s -w " [%{http_code}]\n" http://127.0.0.1:8001/ready http://127.0.0.1:8002/ready http://127.0.0.1:8003/ready http://127.0.0.1:8004/ready
curl -s -o /dev/null -w "  frontend :5173 -> HTTP=%{http_code} bytes=%{size_download}\n" http://localhost:5173/
curl -s -o /dev/null -w "  proxy    :5173/mesas-api -> HTTP=%{http_code} type=%{content_type}\n" http://localhost:5173/mesas-api/api/v1/mesas
TOKEN=$(tok login '{"email":"gm@tfinder.dev","password":"DemoTFinder123!"}'); TOKEN_ADMIN=$(tok login '{"email":"admin@tfinder.dev","password":"DemoTFinder123!"}'); TOKEN_A=$(alta racedemo_a@tfinder.dev "Race A"); TOKEN_B=$(alta racedemo_b@tfinder.dev "Race B"); echo "  TOKEN=${#TOKEN} TOKEN_ADMIN=${#TOKEN_ADMIN} TOKEN_A=${#TOKEN_A} TOKEN_B=${#TOKEN_B}"
curl -s localhost:8001/api/v1/auth/me -H "Authorization: Bearer $TOKEN"; echo
curl -s -o /dev/null -w "  8002 -> %{http_code}\n" localhost:8002/api/v1/builds/1 -H "Authorization: Bearer $TOKEN"
curl -s -o /dev/null -w "  8003 -> %{http_code}\n" localhost:8003/api/v1/feed -H "Authorization: Bearer $TOKEN"
curl -s -o /dev/null -w "  8004 -> %{http_code}\n" localhost:8004/api/v1/notificaciones -H "Authorization: Bearer $TOKEN"
#  lo que tiene que verse: los cuatro 200, :5173 en 200, y TOKEN=243 TOKEN_ADMIN=251 TOKEN_A=260 TOKEN_B=260

# --- BLOQUE 2: estado de las mesas y topologia (20 s) ---------------------------
curl -s localhost:8001/api/v1/mesas | jq -r '.[] | "  mesa \(.id): \(.nombre)  \(.jugadores_actuales)/\(.jugadores_max)  \(.estado)"'
printf "  mesa 1 -> "; curl -s localhost:8001/api/v1/mesas/1/miembros -H "Authorization: Bearer $TOKEN" | jq -r '[.miembros[].email] | join(", ")'
printf "  mesa 2 -> "; curl -s localhost:8001/api/v1/mesas/2/miembros -H "Authorization: Bearer $TOKEN" | jq -r '[.miembros[].email] | join(", ")'
cuenta jugador3@tfinder.dev; cuenta gm@tfinder.dev
docker compose exec -T rabbitmq rabbitmqctl list_queues -p / name durable messages_ready consumers 2>/dev/null | grep -vE 'Timeout|Listing queues|^$'
docker compose exec -T rabbitmq rabbitmqctl list_bindings -p / source_name destination_name routing_key 2>/dev/null | grep -E '^tfinder'
#  lo que tiene que verse: mesa 1 en 2/4, mesa 2 en 3/4, jugador3@=5, gm@=2, dlq.general con 0 consumidores

# --- BLOQUE 3: checklist antes de hablar (30 s) ---------------------------------
curl -s "localhost:8004/api/v1/dlq?limite=1" -H "Authorization: Bearer $TOKEN_ADMIN"; echo
curl -s -o /dev/null -w "  token -> %{http_code}\n" localhost:8001/api/v1/auth/me -H "Authorization: Bearer $TOKEN"
curl -s -o /dev/null -w "  frontend -> %{http_code}\n" http://localhost:5173/
#  la cola de fallidos tiene que dar []

# --- SLIDE 8: las cuatro bases (15 s) -------------------------------------------
docker compose exec -T postgres psql -U tfinder -d postgres -c "\l" | grep -E " (mesas|builds|feed|notif)_db " | awk '{print $1}'
#  builds_db / feed_db / mesas_db / notif_db.  El -d postgres es obligatorio.

# --- SLIDE 9: las dos barreras de concurrencia (45 s) ----------------------------
curl -s localhost:8001/api/v1/mesas/2 | jq -r '"  antes: \(.nombre) \(.jugadores_actuales)/\(.jugadores_max)"'
curl -s -o /tmp/a.json -w '%{http_code}' -X POST localhost:8001/api/v1/mesas/2/solicitudes -H "Authorization: Bearer $TOKEN_A" -H "Idempotency-Key: keyA" -H 'Content-Type: application/json' -d '{}' > /tmp/a.code & curl -s -o /tmp/b.json -w '%{http_code}' -X POST localhost:8001/api/v1/mesas/2/solicitudes -H "Authorization: Bearer $TOKEN_B" -H "Idempotency-Key: keyB" -H 'Content-Type: application/json' -d '{}' > /tmp/b.code & wait
echo "  A -> HTTP $(cat /tmp/a.code): $(jq -c . /tmp/a.json)"; echo "  B -> HTTP $(cat /tmp/b.code): $(jq -c . /tmp/b.json)"; curl -s localhost:8001/api/v1/mesas/2 | jq -r '"  despues: \(.jugadores_actuales)/\(.jugadores_max)"'
#  --> un 201 y un 409, mesa en 4/4. CUAL GANA ES ALEATORIO. El 409 es el lock de Redis.
curl -s -w "  [%{http_code}]\n" -X POST localhost:8001/api/v1/mesas/2/solicitudes -H "Authorization: Bearer $TOKEN" -H "Idempotency-Key: en-serie-$(date +%s)" -H 'Content-Type: application/json' -d '{}'
#  --> {"error":{"codigo":409,"mensaje":"Cupo agotado: no quedan vacantes"}}  <- barrera 2, PostgreSQL
echo -n "  reservas colgadas: "; docker compose exec -T redis redis-cli --scan --pattern 'reserva:*' | tr '\n' ' '; echo "(vacio = liberada)"

# --- SLIDE 10: invalidacion explicita del cache (25 s) ---------------------------
docker compose exec -T redis redis-cli DEL cache:mesas:list > /dev/null; curl -s -o /dev/null localhost:8001/api/v1/mesas; docker compose exec -T redis redis-cli --scan --pattern 'cache:*'; docker compose exec -T redis redis-cli TTL cache:mesas:list
#  --> cache:mesas:list   y   60
curl -s -o /dev/null -X POST localhost:8001/api/v1/mesas -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' -d '{"nombre":"Mesa Demo Para Invalidar","sistema":"PF1e","jugadores_max":4}'
echo -n "  cache tras escribir: "; docker compose exec -T redis redis-cli --scan --pattern 'cache:*' | tr '\n' ' '; echo "(vacio = se borro)"

# --- SLIDE 11: topologia (20 s) --------------------------------------------------
docker compose exec -T rabbitmq rabbitmqctl list_queues -p / name durable messages_ready consumers 2>/dev/null | grep -vE 'Timeout|Listing queues|^$'
docker compose exec -T rabbitmq rabbitmqctl list_bindings -p / source_name destination_name routing_key 2>/dev/null | grep -E '^tfinder'
#  tres colas con 1 consumidor cada una, y dlq.general con 0

# --- SLIDE 11: PDF asincrono de punta a punta (45 s) -----------------------------
DOC=$(curl -s -X POST localhost:8002/api/v1/builds/1/pdf -H "Authorization: Bearer $TOKEN" | jq -r '.documento_id // ("ERROR: " + (.error.mensaje // "?"))'); echo "  documento_id=$DOC"; sleep 2
curl -s "localhost:8004/api/v1/pdf/$DOC/estado" -H "Authorization: Bearer $TOKEN"; echo
curl -s -o doc.pdf -w "  HTTP=%{http_code} type=%{content_type} bytes=%{size_download}\n" "localhost:8004/api/v1/pdf/$DOC" -H "Authorization: Bearer $TOKEN"; file doc.pdf
#  Si el estado todavia no es "listo", repetir la linea anterior hasta que lo sea.
#  --> HTTP=200 type=application/pdf bytes=1257 / PDF document, 1 page(s)

# --- SLIDE 12: modo degradado (25 s) ---------------------------------------------
docker compose pause redis > /dev/null 2>&1; curl -s -w "  [%{http_code}]\n" localhost:8004/ready; curl -s -w "  [%{http_code}]\n" localhost:8004/live; docker compose unpause redis > /dev/null 2>&1
#  --> {"estado":"degradado",...,"redis":false,...}  [503]  y  {"estado":"vivo"}  [200]
#  El unpause va en la MISMA linea: si se olvida, todo lo que sigue da 401.

# --- SLIDE 12: correlacion de punta a punta (30 s) -------------------------------
CID="demo-corr-$(date +%H%M%S)"; curl -s -o /dev/null -w "  solicitud -> %{http_code}\n" -X POST localhost:8001/api/v1/mesas/3/solicitudes -H "Authorization: Bearer $TOKEN" -H "X-Correlation-Id: $CID" -H "Idempotency-Key: k-$CID" -H 'Content-Type: application/json' -d '{}'
docker compose logs mesas-api --no-log-prefix 2>/dev/null | grep "$CID" | tail -2
sleep 2; curl -s "localhost:8004/api/v1/event-log?correlation_id=$CID" -H "Authorization: Bearer $TOKEN_ADMIN" | jq -c .
#  Usa la MESA 3 para no robarle el cupo a la slide 16.

#  --> los cuatro: {"error": {"codigo": 404, "mensaje": "Recurso no encontrado"}}  [404]

KEY="idem-demo-$(date +%s)"; echo "  Idempotency-Key=$KEY"
curl -s -w "  [%{http_code}]\n" -X POST localhost:8001/api/v1/mesas/3/solicitudes -H "Authorization: Bearer $TOKEN" -H "Idempotency-Key: $KEY" -H 'Content-Type: application/json' -d '{}'
curl -s -w "  [%{http_code}]\n" -X POST localhost:8001/api/v1/mesas/3/solicitudes -H "Authorization: Bearer $TOKEN" -H "Idempotency-Key: $KEY" -H 'Content-Type: application/json' -d '{}'
curl -s -w "  [%{http_code}]\n" -X POST localhost:8001/api/v1/mesas/3/solicitudes -H "Authorization: Bearer $TOKEN" -H "Idempotency-Key: ${KEY}-bis" -H 'Content-Type: application/json' -d '{}'
#  --> la 1 y la 2 devuelven el MISMO id y la MISMA fecha. La 3 crea una fila nueva.
